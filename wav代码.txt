def parse_label_from_filename(fname: str):
    base = os.path.basename(fname)
    m = re.match(r"^\s*(\d)", base)
    return int(m.group(1)) if m else None


def read_wav_mono(path):
    sr, x = wavfile.read(path)

    if x.dtype == np.int16:
        x = x.astype(np.float32) / 32768.0
    elif x.dtype == np.int32:
        x = x.astype(np.float32) / 2147483648.0
    elif x.dtype == np.uint8:
        x = (x.astype(np.float32) - 128.0) / 128.0
    else:
        x = x.astype(np.float32)

    if x.ndim == 2:
        x = x.mean(axis=1)

    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    return sr, x

def write_wav_int16(path, sr, x_float):
    x = np.asarray(x_float, dtype=np.float32)
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    x = np.clip(x, -1.0, 1.0)
    x_int16 = (x * 32767.0).astype(np.int16)
    os.makedirs(os.path.dirname(path), exist_ok=True)
    wavfile.write(path, sr, x_int16)

def normalize_proportional(signal):
    signal = np.asarray(signal, dtype=float)
    min_val = np.min(signal)
    max_val = np.max(signal)
    if max_val == min_val:
        return np.ones_like(signal) * 0.5
    return 0.01 + 0.99 * (signal - min_val) / (max_val - min_val)


def load_wav_dataset_from_root(root_dir, mode="clean"):
    data_by_label = defaultdict(list)

    for file in sorted(os.listdir(root_dir)):
        if not file.lower().endswith(".wav"):
            continue

        label = parse_label_from_filename(file)
        if label is None:
            continue

        full_path = os.path.join(root_dir, file)
        _, sig = read_wav_mono(full_path)
        if sig is None or len(sig) == 0:
            continue

       
        sig = normalize_proportional(sig)
        data_by_label[int(label)].append(sig)

    return data_by_label


def spectral_subtraction_denoise(
    x, sr,
    nperseg=512, noverlap=None,
    noise_sec=0.2,
    alpha=1.0,
    beta=0.02,
    smooth_noise=0.0,
):
   
    x = np.asarray(x, dtype=np.float32)
    if x.size < 8:
        return x

    if noverlap is None:
        noverlap = nperseg // 2

    hop = nperseg - noverlap
    if hop <= 0:
        noverlap = nperseg // 2
        hop = nperseg - noverlap

    f, t, Z = stft(
        x, fs=sr, window="hann",
        nperseg=nperseg, noverlap=noverlap,
        boundary="zeros", padded=True
    )

    mag = np.abs(Z)
    phase = np.angle(Z)

    noise_frames = max(1, int(noise_sec * sr / hop))
    noise_frames = min(noise_frames, mag.shape[1])
    noise_mag = np.mean(mag[:, :noise_frames], axis=1, keepdims=True)

    if smooth_noise and 0 < smooth_noise < 1.0:
        a = float(smooth_noise)
        D = noise_mag.copy()
        for k in range(1, D.shape[0]):
            D[k] = a * D[k-1] + (1-a) * D[k]
        noise_mag = D

    mag_clean = mag - alpha * noise_mag
    mag_floor = beta * mag
    mag_clean = np.maximum(mag_clean, mag_floor)

    Z_clean = mag_clean * np.exp(1j * phase)

    _, x_rec = istft(
        Z_clean, fs=sr, window="hann",
        nperseg=nperseg, noverlap=noverlap,
        input_onesided=True, boundary=True
    )

    if x_rec.size > x.size:
        x_rec = x_rec[:x.size]
    elif x_rec.size < x.size:
        x_rec = np.pad(x_rec, (0, x.size - x_rec.size), mode="edge")

    x_rec = np.nan_to_num(x_rec, nan=0.0, posinf=0.0, neginf=0.0)
    mx = np.max(np.abs(x_rec)) + 1e-9
    if mx > 1.0:
        x_rec = x_rec / mx

    return x_rec.astype(np.float32)

def denoise_wav_folder_spectral_subtraction(
    noisy_root, denoised_root,
    overwrite=True,               
    nperseg=512,
    noise_sec=0.2,
    alpha=1.0,
    beta=0.02,
    smooth_noise=0.0,
    print_every=50
):
    os.makedirs(denoised_root, exist_ok=True)
    wavs = [f for f in sorted(os.listdir(noisy_root)) if f.lower().endswith(".wav")]

    processed, skipped = 0, 0
    for idx, fn in enumerate(wavs, 1):
        in_path = os.path.join(noisy_root, fn)
        out_path = os.path.join(denoised_root, fn)

        if (not overwrite) and os.path.exists(out_path):
            skipped += 1
            continue

        sr, x = read_wav_mono(in_path)
        x_dn = spectral_subtraction_denoise(
            x, sr,
            nperseg=nperseg,
            noverlap=nperseg // 2,
            noise_sec=noise_sec,
            alpha=alpha,
            beta=beta,
            smooth_noise=smooth_noise
        )
        write_wav_int16(out_path, sr, x_dn)
        processed += 1

        if print_every and (idx % print_every == 0):
            print(f"[SpectralSub] {idx}/{len(wavs)} processed={processed}, skipped={skipped}")

    print(f"[SpectralSub] 完成：total={len(wavs)} processed={processed} skipped={skipped}")
    print("输出目录：", os.path.abspath(denoised_root))


def robust_norm_01(x, eps=1e-6, z_clip=10.0):
    x = np.asarray(x, dtype=float)
    med = np.median(x, axis=0)
    mad = np.median(np.abs(x - med), axis=0) + eps
    z = (x - med) / mad
    z = np.clip(z, -z_clip, z_clip)  
    return 1.0 / (1.0 + np.exp(-z))

def segment_stats_with_peakcount(signal, num_segments=40, peak_prom=0.15, min_len=64):
    signal = np.asarray(signal, dtype=float)
    n = len(signal)

    target_min = int(num_segments * min_len)
    if n < target_min and n > 1:
        signal = resample(signal, target_min).astype(float)
        n = len(signal)

    if n < 2:
        signal = np.pad(signal, (0, max(1, target_min - n)), mode="edge")
        n = len(signal)

    seg_len = n // num_segments
    if seg_len < 2:
        signal = resample(signal, num_segments * 2).astype(float)
        n = len(signal)
        seg_len = max(2, n // num_segments)

    feats = []
    for i in range(num_segments):
        start = i * seg_len
        end = (i + 1) * seg_len if i < num_segments - 1 else n
        seg = signal[start:end]
        if seg.size < 2:
            seg = np.pad(seg, (0, 2 - seg.size), mode="edge")

        mean = float(np.mean(seg))
        peak = float(np.max(seg))
        amp  = float(np.max(seg) - np.min(seg))

        peaks, _ = find_peaks(seg, prominence=peak_prom)
        peak_count = float(len(peaks))

        feats.append([mean, peak, amp, peak_count])

    return np.array(feats, dtype=float)

def threshold_count_encode(feats_01, thr=(0.2, 0.4, 0.6, 0.8)):
    thr = np.array(thr, dtype=float)
    counts = (feats_01[..., None] > thr[None, None, :]).sum(axis=-1).astype(np.float32)
    counts = counts / max(len(thr), 1)
    return counts

def encode_signal_to_vector(signal,
                            num_segments=40,
                            peak_prom=0.15,
                            thr=(0.2, 0.4, 0.6, 0.8),
                            peakcount_clip=10,
                            min_len=64):
    feats = segment_stats_with_peakcount(
        signal,
        num_segments=num_segments,
        peak_prom=peak_prom,
        min_len=min_len
    )
    feats_01 = robust_norm_01(feats[:, :3])  # (S,3)
    pc = np.clip(feats[:, 3], 0, peakcount_clip) / float(peakcount_clip)  # (S,)
    counts = threshold_count_encode(feats_01, thr=thr)  # (S,3)

    vec = np.concatenate([counts.reshape(-1), pc.reshape(-1)], axis=0).astype(np.float32)
    return vec

def build_vector_dataset_train_val(data_by_label, train_ratio=0.9, num_segments=40,
                                   peak_prom=0.15, thr=(0.2,0.4,0.6,0.8),
                                   peakcount_clip=10, min_len=64):
    X_train, y_train, X_val, y_val = [], [], [], []
    for label in sorted(data_by_label.keys()):
        signals = data_by_label[label]
        if len(signals) < 2:
            continue
        tr, va = train_test_split(signals, train_size=train_ratio, random_state=seed)
        for s in tr:
            X_train.append(encode_signal_to_vector(s, num_segments, peak_prom, thr, peakcount_clip, min_len))
            y_train.append(label)
        for s in va:
            X_val.append(encode_signal_to_vector(s, num_segments, peak_prom, thr, peakcount_clip, min_len))
            y_val.append(label)
    return np.array(X_train, dtype=np.float32), np.array(y_train), np.array(X_val, dtype=np.float32), np.array(y_val)

def build_vector_dataset_all(data_by_label, num_segments=40,
                            peak_prom=0.15, thr=(0.2,0.4,0.6,0.8),
                            peakcount_clip=10, min_len=64):
    X, y = [], []
    for label in sorted(data_by_label.keys()):
        for s in data_by_label[label]:
            X.append(encode_signal_to_vector(s, num_segments, peak_prom, thr, peakcount_clip, min_len))
            y.append(label)
    return np.array(X, dtype=np.float32), np.array(y)


def repeat_current(x: torch.Tensor, T=50, noise_std=0.0):
    cur = x.unsqueeze(0).repeat(T, 1, 1)
    if noise_std > 0:
        cur = cur + noise_std * torch.randn_like(cur)
    return cur


class ANNSNN(nn.Module):
    def __init__(self, input_dim, hidden1=256, hidden2=128, num_classes=10,
                 beta=0.9, threshold=1.0, dropout=0.2):
        super().__init__()
        self.fc1 = nn.Linear(input_dim, hidden1)
        self.ln1 = nn.LayerNorm(hidden1)
        self.lif1 = snn.Leaky(beta=beta, threshold=threshold)
        self.dp1 = nn.Dropout(dropout)

        self.fc2 = nn.Linear(hidden1, hidden2)
        self.ln2 = nn.LayerNorm(hidden2)
        self.lif2 = snn.Leaky(beta=beta, threshold=threshold)
        self.dp2 = nn.Dropout(dropout)

        self.fc3 = nn.Linear(hidden2, num_classes)
        self.lif_out = snn.Leaky(beta=beta, threshold=threshold)

    def forward(self, x_vec, T=50, noise_std=0.0):
        cur_in = repeat_current(x_vec, T=T, noise_std=noise_std)

        mem1 = self.lif1.init_leaky()
        mem2 = self.lif2.init_leaky()
        memo = self.lif_out.init_leaky()

        mem_out_rec, spk_out_rec = [], []
        for t in range(T):
            x = cur_in[t]

            cur1 = self.fc1(x)
            cur1 = self.ln1(cur1)
            cur1 = torch.relu(cur1)
            spk1, mem1 = self.lif1(cur1, mem1)
            spk1 = self.dp1(spk1)

            cur2 = self.fc2(spk1)
            cur2 = self.ln2(cur2)
            cur2 = torch.relu(cur2)
            spk2, mem2 = self.lif2(cur2, mem2)
            spk2 = self.dp2(spk2)

            cur3 = self.fc3(spk2)
            spk3, memo = self.lif_out(cur3, memo)

            mem_out_rec.append(memo)
            spk_out_rec.append(spk3)

        mem_out = torch.stack(mem_out_rec, dim=0)
        spk_out = torch.stack(spk_out_rec, dim=0)
        return mem_out, spk_out


def train_on_clean_ann_snn(X_train, y_train, X_val, y_val,
                           epochs=60, batch_size=64, lr=1e-3, weight_decay=2e-3,
                           T=70, beta=0.9, threshold=0.8, dropout=0.2,
                           noise_std=0.01, grad_clip=1.0,
                           avg_last_k=20, vote_k=1):
    pin = (device.type == "cuda")

    train_ds = TensorDataset(torch.tensor(X_train), torch.tensor(y_train).long())
    val_ds   = TensorDataset(torch.tensor(X_val),   torch.tensor(y_val).long())
    train_loader = DataLoader(train_ds, batch_size=batch_size, shuffle=True, pin_memory=pin)
    val_loader   = DataLoader(val_ds,   batch_size=batch_size, shuffle=False, pin_memory=pin)

    net = ANNSNN(
        input_dim=X_train.shape[1],
        hidden1=256,
        hidden2=128,
        num_classes=10,
        beta=beta,
        threshold=threshold,
        dropout=dropout
    ).to(device)

    opt = torch.optim.AdamW(net.parameters(), lr=lr, weight_decay=weight_decay)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=epochs)
    loss_fn = nn.CrossEntropyLoss()

    best_acc = -1.0
    best_state = None
    loss_list, acc_list = [], []

    for ep in range(1, epochs + 1):
        net.train()
        total_loss = 0.0

        for xb, yb in train_loader:
            xb = xb.to(device, non_blocking=pin)
            yb = yb.to(device, non_blocking=pin)

            mem_out, _ = net(xb, T=T, noise_std=noise_std)
            logits = mem_out[-avg_last_k:].mean(dim=0)
            loss = loss_fn(logits, yb)

            opt.zero_grad()
            loss.backward()
            if grad_clip is not None and grad_clip > 0:
                torch.nn.utils.clip_grad_norm_(net.parameters(), max_norm=grad_clip)
            opt.step()
            total_loss += loss.item()

        scheduler.step()

        net.eval()
        correct, total = 0, 0
        with torch.no_grad():
            for xb, yb in val_loader:
                xb = xb.to(device, non_blocking=pin)
                yb = yb.to(device, non_blocking=pin)

                logits_sum = 0.0
                for _ in range(max(1, vote_k)):
                    mem_out, _ = net(xb, T=T, noise_std=0.0)
                    logits_sum = logits_sum + mem_out[-avg_last_k:].mean(dim=0)

                pred = (logits_sum / max(1, vote_k)).argmax(dim=1)
                correct += (pred == yb).sum().item()
                total += yb.size(0)

        avg_loss = total_loss / max(1, len(train_loader))
        acc = correct / max(1, total)
        loss_list.append(avg_loss)
        acc_list.append(acc)

        if acc > best_acc:
            best_acc = acc
            best_state = {k: v.detach().cpu() for k, v in net.state_dict().items()}

        cur_lr = scheduler.get_last_lr()[0]
        print(f"Epoch {ep:02d}: Loss={avg_loss:.4f}, ValAcc={acc:.4f}, Best={best_acc:.4f}, LR={cur_lr:.6f}")

    if best_state is not None:
        net.load_state_dict(best_state)

    return net, loss_list, acc_list

def evaluate_on_dataset_ann_snn(net, X, y, title="Test", T=70, avg_last_k=20, vote_k=5):
    pin = (device.type == "cuda")
    ds = TensorDataset(torch.tensor(X), torch.tensor(y).long())
    loader = DataLoader(ds, batch_size=128, shuffle=False, pin_memory=pin)

    net.eval()
    y_true, y_pred = [], []
    with torch.no_grad():
        for xb, yb in loader:
            xb = xb.to(device, non_blocking=pin)

            logits_sum = 0.0
            for _ in range(max(1, vote_k)):
                mem_out, _ = net(xb, T=T, noise_std=0.0)
                logits_sum = logits_sum + mem_out[-avg_last_k:].mean(dim=0)

            pred = (logits_sum / max(1, vote_k)).argmax(dim=1).cpu().numpy()
            y_true.extend(yb.numpy())
            y_pred.extend(pred)

    y_true = np.array(y_true)
    y_pred = np.array(y_pred)
    acc = float((y_true == y_pred).mean())

    print(f"\n[{title}] Accuracy: {acc:.4f}\n")
    print(classification_report(y_true, y_pred, digits=4, zero_division=0))

    cm = confusion_matrix(y_true, y_pred)
    ConfusionMatrixDisplay(confusion_matrix=cm).plot(cmap="Purples")
    plt.title(f"Confusion Matrix - {title}")
    plt.show()

    return y_true, y_pred, acc, cm

    plt.legend()

    plt.show()
